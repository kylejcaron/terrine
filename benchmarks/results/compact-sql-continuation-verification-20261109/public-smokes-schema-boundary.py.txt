def digest(data):
    return hashlib.sha256(data).hexdigest()


def save_arrow(name, table):
    sink = pa.BufferOutputStream()
    with pa.ipc.new_stream(sink, table.schema) as writer:
        writer.write_table(table)
    raw = sink.getvalue().to_pybytes()
    (OUT / name).write_bytes(raw)
    return {'rows': table.num_rows, 'schema': str(table.schema), 'arrow_stream_sha256': digest(raw)}


def read_arrow(path):
    with pa.memory_map(str(path), 'r') as source:
        return pa.ipc.open_stream(source).read_all()


def daily(con, unit, label):
    day = 86400 * (10**6 if unit == 'us' else 10**9)
    ticks = [-day - 1, -day, -1, 0, day - 1, day]
    users = pa.table({'user_id': ['a', 'empty']})
    events = pa.table({
        'user_id': ['a'] * 6,
        'occurred_at': pa.array(ticks, type=pa.timestamp(unit)),
        'qty': pa.array([100, 2, 3, 4, 5, 100], type=pa.int64()),
        'amount': pa.array([Decimal(x) for x in ('100', '2', '3', '4', '5', '100')], type=pa.decimal128(12, 2)),
        'weight': [100.0, 2.0, 3.0, 4.0, 5.0, 100.0],
        'light': pa.array([1.0] * 6, type=pa.float32()),
    })
    con.create_table('users', users, overwrite=True)
    con.create_table('events', events, overwrite=True)
    engine = 'postgres' if label == 'postgres-us' else 'duckdb'
    if engine == 'postgres':
        con.raw_sql("SET TimeZone = 'America/New_York'").close()
    else:
        con.con.execute("SET TimeZone = 'America/New_York'")
    with tempfile.TemporaryDirectory() as directory:
        catalog = Path(directory) / 'catalog.yaml'
        catalog.write_text(pg_windows._CATALOG.format(dialect=engine))
        model = tr.Model.from_catalog(catalog, con=con, units={'user': tr.UnitSource('users', 'user_id')}, readiness={'orders': tr.Coverage('1969-12-30', '1970-01-03')})
        design = tr.Design('user', tr.Schedule(every='1d', start=date(1969, 12, 31), end=date(1970, 1, 1)), outcomes={'counted': tr.agg('orders', window=('0s', '1d')), 'total': tr.agg('spend', window=('0s', '1d'))})
        frame = design.build(model)
        strategies = dict(frame._compiled.strategy)
        assert all(value.startswith('local_day/') for value in strategies.values()), strategies
        result = frame.materialize(budget=tr.Budget(max_rows=10, max_output_bytes=100_000, strict_scan=False))
        table = result.table.sort_by([('unit', 'ascending'), ('t0', 'ascending')])
        rows = table.select(['unit', 'counted', 'total', 'counted__complete', 'total__complete', 'counted__reason', 'total__reason']).to_pylist()
        assert rows == [
            {'unit': 'a', 'counted': 2, 'total': Decimal('5.00'), 'counted__complete': True, 'total__complete': True, 'counted__reason': None, 'total__reason': None},
            {'unit': 'a', 'counted': 2, 'total': Decimal('9.00'), 'counted__complete': True, 'total__complete': True, 'counted__reason': None, 'total__reason': None},
            {'unit': 'empty', 'counted': 0, 'total': Decimal('0.00'), 'counted__complete': True, 'total__complete': True, 'counted__reason': None, 'total__reason': None},
            {'unit': 'empty', 'counted': 0, 'total': Decimal('0.00'), 'counted__complete': True, 'total__complete': True, 'counted__reason': None, 'total__reason': None},
        ], rows
        assert table['t0'].cast(pa.int64()).to_pylist() == [-86_400_000_000, 0, -86_400_000_000, 0]
        zone = con.raw_sql("SELECT current_setting('TimeZone')")
        try:
            assert zone.fetchone()[0] == 'America/New_York'
        finally:
            if engine == 'postgres':
                zone.close()
        return {'boundary': 'half-open two-day schedule; naive source, pre-epoch lower/upper/adjacent ticks; NY caller session unchanged', 'source_unit': unit, 'strategies': strategies, 'public': save_arrow(f'{label}-public.arrow', table), 'rows': str(rows)}


def state():
    prior = ROOT / 'benchmarks/results/compact-sql-task8-state-specific-20261109-attempt2'
    con = ibis.duckdb.connect(threads=1)
    try:
        con.con.execute("SET TimeZone='UTC'")
        workloads.load_tables(con, test_cost_budget._CONFIG)
        model = workloads.cost_model(con, test_cost_budget._CONFIG)
        case = workloads.recipe_cases(test_cost_budget._CONFIG)['state_panel']
        frame = case.design.build(model)
        consumer_chars = len(ibis.to_sql(frame.expr))
        assert consumer_chars <= 30_000
        with compare_sql._capture_study_sql() as statements:
            result = frame.materialize(budget=tr.Budget(max_rows=25_000, max_output_bytes=20_000_000, strict_scan=False))
        table = result.table.sort_by([('unit', 'ascending'), ('t0', 'ascending')])
        metadata = result.to_metadata()
        dynamic = {name: metadata['provenance']['execution_snapshot'].pop(name) for name in ('snapshot_id', 'input_binding_id')}
        for mode in ('original_union_fallback', 'compact_query'):
            before = read_arrow(prior / mode / 'public-full.arrow').sort_by([('unit', 'ascending'), ('t0', 'ascending')])
            table = table.select(before.column_names)
            assert table.schema.serialize().to_pybytes() == before.schema.serialize().to_pybytes(), mode
            assert table.equals(before, check_metadata=True), mode
            assert metadata == json.loads((prior / mode / 'invariant-metadata.json').read_text()), mode
        assert table.num_rows == 20_000 and table.num_columns == 20
        assert table.schema.field('gap').type == pa.duration('us')
        reference = compare_sql._execute(con, case.reference).sort_by([('unit', 'ascending'), ('t0', 'ascending')])
        expected = reference.to_pylist()
        projection = table.select(list(case.key + case.values))
        compare_sql._assert_schema(projection, reference)
        compare_sql._assert_rows(projection, expected)
        controls = {}
        for name in ('missing', 'duplicate', 'changed_value', 'changed_type'):
            candidate = read_arrow(prior / f'control-{name}.arrow')
            try:
                if name == 'changed_type':
                    compare_sql._assert_schema(candidate, reference)
                else:
                    compare_sql._assert_rows(candidate, expected)
            except AssertionError:
                controls[name] = True
            else:
                raise AssertionError(name)
        full = con.compile(frame.expr)
        (OUT / 'state-full-expression.sql').write_text(full + '\n')
        for index, sql in enumerate(statements):
            (OUT / f'state-public-study-{index}.sql').write_text(sql + '\n')
        (OUT / 'state-invariant-metadata.json').write_text(json.dumps(metadata, indent=2) + '\n')
        return {'consumer_formatted_chars': consumer_chars, 'schema_values_support_columnmeta_invariant_provenance_match_both_pre7_modes': True, 'dynamic_snapshot_identities_recorded_not_compared': dynamic, 'reference_parity': True, 'controls': controls, 'public': save_arrow('state-public-full.arrow', table), 'public_study_sql': {'statements': len(statements), 'bytes': sum(len(sql.encode()) for sql in statements), 'excludes': 'metadata/admission probes, transaction control, catalog introspection'}}
    finally:
        con.disconnect()


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--postgres', action='store_true')
    parser.add_argument('--daily-only', action='store_true')
    args = parser.parse_args()
    assert Path(tr.__file__).resolve().is_relative_to(ROOT)
    evidence = {'duckdb': duckdb.__version__, 'ibis': ibis.__version__, 'module': tr.__file__}
    if args.postgres:
        schema = 'terrine_daily_' + uuid.uuid4().hex[:12]
        admin = connect_postgres()
        admin.raw_sql(f'CREATE SCHEMA "{schema}"').close()
        con = connect_postgres(schema=schema)
        try:
            evidence['postgres-us'] = daily(con, 'us', 'postgres-us')
        finally:
            con.disconnect()
            admin.raw_sql(f'DROP SCHEMA "{schema}" CASCADE').close()
            admin.disconnect()
        name = 'postgres-public-daily.json'
    else:
        for unit in ('us', 'ns'):
            con = ibis.duckdb.connect(threads=1)
            try:
                evidence[f'duckdb-{unit}'] = daily(con, unit, f'duckdb-{duckdb.__version__}-{unit}')
            finally:
                con.disconnect()
        if not args.daily_only:
            evidence['state'] = state()
        name = f'duckdb-{duckdb.__version__}-public-smokes.json'
    (OUT / name).write_text(json.dumps(evidence, indent=2) + '\n')
    print(json.dumps(evidence, indent=2))


if __name__ == '__main__':
    main()
